{{- /* Tables scroll inside their own box on narrow screens. A table of figures keeps each cell on one line; a table with sentences in it (class "prose") lets them wrap. */ -}} {{- $prose := false -}} {{- range .TBody }}{{ range . }}{{ if gt (len (.Text | plainify)) 48 }}{{ $prose = true }}{{ end }}{{ end }}{{ end -}} {{- /* A table whose header cells are all empty has no header. */ -}} {{- $head := false -}} {{- range .THead }}{{ range . }}{{ if .Text | plainify | strings.TrimSpace }}{{ $head = true }}{{ end }}{{ end }}{{ end -}}
{{- if $head }} {{- range .THead }} {{- range . }} {{ .Text }} {{- end }} {{- end }} {{- end }} {{- range .TBody }} {{- range . }} {{ .Text }} {{- end }} {{- end }}
